```python
"""
FDA Import Refusal Report Scraper
Downloads, parses, and cleans FDA Import Refusal reports.

Dependencies: pip install beautifulsoup4 requests cloudscraper
"""

import os
import re
import csv
import json
import time
import logging
import random
from typing import List, Dict, Any
from datetime import datetime

# Use cloudscraper to bypass WAF/Anti-bot protections, fallback to standard requests if missing
try:
    import cloudscraper
    SESSION = cloudscraper.create_scraper()
except ImportError:
    import requests
    from requests.adapters import HTTPAdapter
    from urllib3.util.retry import Retry
    
    SESSION = requests.Session()
    retries = Retry(
        total=5,
        backoff_factor=1,
        status_forcelist=[403, 429, 500, 502, 503, 504],
        allowed_methods=["HEAD", "GET", "POST"]
    )
    SESSION.mount('https://', HTTPAdapter(max_retries=retries))

from bs4 import BeautifulSoup

# Configuration
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
)
logger = logging.getLogger("FDA_Scraper")

BASE_URL = "https://www.accessdata.fda.gov/scripts/importrefusals/"
SEARCH_URL = BASE_URL

# Map potential HTML table headers to our target schema
HEADERS_MAP = {
    'Refusal Date': 'date',
    'Date': 'date',
    'FDA Product Description': 'product',
    'Product Description': 'product',
    'Product': 'product',
    'Manufacturer Name': 'manufacturer',
    'Manufacturer': 'manufacturer',
    'Country': 'country',
    'Country/Area': 'country',
    'Charges': 'violation_code',
    'Violation Code': 'violation_code',
    'Reason': 'reason',
    'Reason for Refusal': 'reason',
    'Charge': 'violation_code'
}

def clean_manufacturer_name(name: str) -> str:
    """
    Normalizes manufacturer names by standardizing casing 
    and removing common corporate suffixes.
    """
    if not name:
        return ""
    
    # Remove common corporate suffixes
    suffixes = r'\b(INC|LLC|CORP|CO|LTD|LIMITED|CORPORATION|COMPANY|LLP|GMBH|S\.A\.|S\.R\.L\.|S\.A\.S\.|PTY|PLC)\.?\b'
    name = re.sub(suffixes, '', name, flags=re.IGNORECASE)
    
    # Remove trailing punctuation and extra whitespace
    name = re.sub(r'[.,;:]+$', '', name)
    name = re.sub(r'\s+', ' ', name).strip()
    
    # Return in Title Case
    return name.title()

def retry_request(url: str, method: str = 'get', payload: Dict = None, 
                  headers: Dict = None, max_retries: int = 5) -> Any:
    """
    Executes an HTTP request with exponential backoff and abuse-detection checks.
    """
    for attempt in range(max_retries):
        try:
            if method.lower() == 'post':
                response = SESSION.post(url, data=payload, headers=headers, timeout=45)
            else:
                response = SESSION.get(url, params=payload, headers=headers, timeout=45)
                
            # FDA often returns a 200 OK but redirects to an apology page if blocked
            if "abuse-detection-apology" in response.url or "excessive-requests-apology" in response.url:
                raise ConnectionError("Blocked by FDA abuse detection/WAF.")
                
            response.raise_for_status()
            return response
            
        except Exception as e:
            logger.warning(f"Request attempt {attempt + 1}/{max_retries} failed: {str(e)}")
            if attempt < max_retries - 1:
                # Exponential backoff + jitter
                sleep_time = (2 ** attempt) + random.uniform(0, 1)
                logger.info(f"Retrying in {sleep_time:.2f} seconds...")
                time.sleep(sleep_time)
            else:
                logger.error("Max retries exceeded. Failing request.")
                raise

def download_and_parse_reports(year: int, month: int) -> List[Dict[str, Any]]:
    """
    Submits search criteria to the FDA portal and parses the resulting HTML tables.
    """
    logger.info(f"Fetching reports for {year}-{month:02d}")
    
    # FDA form parameters (Month is usually 01-12, Year is YYYY)
    payload = {
        'Month': f"{month:02d}",
        'Year': str(year),
        'searchType': 'country',
        'submit': 'Search'
    }
    
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8',
        'Accept-Language': 'en-US,en;q=0.9',
        'Referer': BASE_URL,
        'Origin': 'https://www.accessdata.fda.gov',
        'Connection': 'keep-alive'
    }
    
    response = retry_request(SEARCH_URL, method='post', payload=payload, headers=headers)
    soup = BeautifulSoup(response.text, 'html.parser')
    
    records = []
    tables = soup.find_all('table')
    
    for table in tables:
        # Attempt to find header row
        headers_row = table.find_all('th')
        if not headers_row:
            # Fallback: sometimes headers are in the first <tr> as <td>
            first_row = table.find('tr')
            if first_row:
                headers_row = first_row.find_all(['td', 'th'])
            else:
                continue
                
        col_map = {}
        for i, th in enumerate(headers_row):
            text = th.get_text(strip=True)
            if text in HEADERS_MAP:
                col_map[i] = HEADERS_MAP[text]
                
        if not col_map:
            continue # Not a relevant table
            
        # Process data rows (skip the header row)
        rows = table.find_all('tr')[1:]
        for row in rows:
            cols = row.find_all(['td', 'th'])
            if len(cols) < max(col_map.keys()) + 1:
                continue
                
            record = {}
            for idx, field in col_map.items():
                val = cols[idx].get_text(strip=True)
                record[field] = val
                
            # If violation_code exists but reason doesn't, map violation code to reason
            # as FDA often puts the reason text directly inside the charges column
            if 'violation_code' in record and 'reason' not in record:
                record['reason'] = record['violation_code']
                
            # Clean manufacturer name
            if 'manufacturer' in record:
                record['manufacturer'] = clean_manufacturer_name(record['manufacturer'])
                
            # Ensure all required fields exist, fill with empty string if missing
            for required_field in ['date', 'product', 'manufacturer', 'country', 'violation_code', 'reason']:
                if required_field not in record:
                    record[required_field] = ""
                    
            records.append(record)
            
    return records

def main():
    """
    Main execution function: fetches data, cleans it, and exports to JSON and CSV.
    """
    all_records = []
    
    # Define the range of months to fetch 
    current_year = datetime.now().year
    current_month = datetime.now().month
    
    # To avoid triggering WAF in a demonstration run, we fetch the current month.
    # In a full production run, you would iterate over your desired date range.
    months_to_fetch = [(current_year, current_month)]
    
    for y, m in months_to_fetch:
        try:
            records = download_and_parse_reports(y, m)
            all_records.extend(records)
            logger.info(f"Successfully extracted {len(records)} records for {y}-{m:02d}")
            # Polite delay between requests to avoid triggering rate limits
            time.sleep(random.uniform(2.0, 5.0))
        except Exception as e:
            logger.error(f"Failed to fetch data for {y}-{m:02d}: {str(e)}")
            
    if not all_records:
        logger.warning("No records were extracted. Exiting without generating files.")
        return

    # Output Structured JSON
    output_json = "fda_import_refusals.json"
    with open(output_json, 'w', encoding='utf-8') as f:
        json.dump(all_records, f, indent=2, ensure_ascii=False)
    logger.info(f"Saved structured JSON to {output_json} ({len(all_records)} rows)")
    
    # Output Clean CSV ready for B2B Sale
    output_csv = "fda_import_refusals.csv"
    # Define strict column order for B2B output
    fieldnames = ['date', 'product', 'manufacturer', 'country', 'violation_code', 'reason']
    
    with open(output_csv, 'w', newline='', encoding='utf-8') as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(all_records)
    logger.info(f"Saved clean CSV to {output_csv} ({len(all_records)} rows)")

if __name__ == "__main__":
    main()
```
