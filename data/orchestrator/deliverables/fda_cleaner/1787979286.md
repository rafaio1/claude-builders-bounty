```python
import requests
import pandas as pd
import json
import csv
import re
import logging
import time
from datetime import datetime
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry
import io

# Configure logging for production monitoring
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[logging.StreamHandler()]
)
logger = logging.getLogger(__name__)

def get_session_with_retries():
    """Creates a requests session with robust retry logic and backoff."""
    session = requests.Session()
    retry_strategy = Retry(
        total=5,
        backoff_factor=2,
        status_forcelist=[429, 500, 502, 503, 504],
        allowed_methods=["HEAD", "GET", "POST"]
    )
    adapter = HTTPAdapter(max_retries=retry_strategy)
    session.mount("http://", adapter)
    session.mount("https://", adapter)
    session.headers.update({
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/115.0.0.0 Safari/537.36",
        "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8",
        "Accept-Language": "en-US,en;q=0.5",
        "Referer": "https://www.accessdata.fda.gov/scripts/importrefusals/"
    })
    return session

def download_reports(session, base_url):
    """Submits the search form to generate and download the HTML table."""
    logger.info(f"Fetching data from: {base_url}")
    
    # We use the current or recent month/year to trigger the HTML table generation
    now = datetime.now()
    month_name = now.strftime("%B")
    year = str(now.year)
    
    # The FDA form typically expects the month name and year.
    payload = {
        "db": "irr", 
        "month": month_name,
        "year": year,
        "search": "Search"
    }
    
    try:
        # First, get the main page to establish cookies/session
        session.get(base_url, timeout=30)
        
        # Post to the search endpoint
        response = session.post(base_url, data=payload, timeout=30)
        
        # If the site strictly uses GET for some parameters or redirects, handle it
        if response.status_code == 405: # Method Not Allowed
             logger.warning("POST not allowed, attempting GET with query parameters.")
             response = session.get(base_url, params=payload, timeout=30)
             
        response.raise_for_status()
        return response.text
        
    except requests.exceptions.RequestException as e:
        logger.error(f"Failed to download reports: {e}")
        raise

def clean_manufacturer(name):
    """Normalizes casing and removes common corporate suffixes."""
    if not isinstance(name, str) or not name.strip():
        return ""
    
    name = name.strip().rstrip('.,;')
    
    # Suffixes to remove
    suffixes = r'\b(inc|incorporated|llc|corp|corporation|company|co|ltd|limited|sa|de|cv|srl|pte|ltda|gmbh|ag|bv|nv|plc)\b[\.]?'
    
    name = re.sub(suffixes, '', name, flags=re.IGNORECASE)
    name = re.sub(r'[\s\.,;]+$', '', name)
    name = re.sub(r'^[\s\.,;]+', '', name)
    name = re.sub(r'\s+', ' ', name)
    
    # Title case for B2B consistency
    return name.strip().title()

def parse_and_clean_data(html_content):
    """Parses HTML tables into structured JSON and cleans manufacturer names."""
    logger.info("Parsing HTML tables...")
    
    try:
        # Use pandas with BeautifulSoup flavor for robust HTML table extraction
        tables = pd.read_html(io.StringIO(html_content), flavor='bs4')
    except ValueError:
        logger.error("No tables found in the HTML content. The site layout may have changed.")
        return pd.DataFrame()

    if not tables:
        logger.warning("Parsed HTML but found no tables.")
        return pd.DataFrame()

    # Search for the table containing the refusal data
    target_df = None
    for df in tables:
        cols_lower = [str(c).lower() for c in df.columns]
        if any('manufacturer' in c for c in cols_lower) or any('product' in c for c in cols_lower) or any('charges' in c for c in cols_lower):
            target_df = df
            break
            
    if target_df is None:
        target_df = max(tables, key=lambda x: x.size)
        logger.warning("Could not find exact header match. Using the largest table found.")

    # Map FDA columns to our required fields
    col_mapping = {}
    for col in target_df.columns:
        col_lower = str(col).lower()
        if 'date' in col_lower or 'refusal date' in col_lower: col_mapping[col] = 'date'
        elif 'product' in col_lower or 'description' in col_lower: col_mapping[col] = 'product'
        elif 'manufacturer' in col_lower or 'firm' in col_lower: col_mapping[col] = 'manufacturer'
        elif 'country' in col_lower: col_mapping[col] = 'country'
        elif 'charge' in col_lower or 'violation' in col_lower or 'code' in col_lower: col_mapping[col] = 'violation_code'
        elif 'reason' in col_lower: col_mapping[col] = 'reason'
        
    target_df.rename(columns=col_mapping, inplace=True)
    
    # If 'violation_code' and 'reason' are combined in 'CHARGES', separate them
    if 'violation_code' in target_df.columns and 'reason' not in target_df.columns:
        # Often charges look like "402(a)(1) - Filthy" or similar
        target_df['reason'] = target_df['violation_code'].astype(str).apply(lambda x: x.split('-', 1)[-1].strip() if '-' in str(x) else "")
        target_df['violation_code'] = target_df['violation_code'].astype(str).apply(lambda x: x.split('-', 1)[0].strip() if '-' in str(x) else x)
    elif 'violation_code' not in target_df.columns:
        target_df['violation_code'] = ""
        
    required_fields = ['date', 'product', 'manufacturer', 'country', 'violation_code', 'reason']
    for field in required_fields:
        if field not in target_df.columns:
            target_df[field] = ""
            
    target_df = target_df[required_fields]
    
    # Clean data
    target_df['manufacturer'] = target_df['manufacturer'].apply(clean_manufacturer)
    
    for col in ['product', 'country', 'violation_code', 'reason', 'date']:
        target_df[col] = target_df[col].astype(str).apply(lambda x: ' '.join(x.split()).strip())
        target_df[col] = target_df[col].replace('nan', '')
        
    # Drop rows that are completely empty
    target_df.dropna(how='all', inplace=True)
    target_df = target_df[target_df['product'] != '']
    
    logger.info(f"Successfully parsed and cleaned {len(target_df)} records.")
    return target_df

def export_data(df, json_filename="fda_import_refusals.json", csv_filename="fda_import_refusals_b2b.csv"):
    """Exports the cleaned DataFrame to JSON and CSV."""
    if df.empty:
        logger.warning("DataFrame is empty. Skipping export.")
        return
        
    # Export to structured JSON
    json_data = df.to_dict(orient='records')
    with open(json_filename, 'w', encoding='utf-8') as f:
        json.dump(json_data, f, indent=4, ensure_ascii=False)
    logger.info(f"Data exported to {json_filename}")
    
    # Export to clean CSV ready for B2B sale
    df.to_csv(csv_filename, index=False, quoting=csv.QUOTE_ALL, encoding='utf-8')
    logger.info(f"Data exported to {csv_filename}")

def main():
    """Main function to run the end-to-end pipeline."""
    base_url = "https://www.accessdata.fda.gov/scripts/importrefusals/"
    
    # Note: The FDA also provides an official API at https://api-datadashboard.fda.gov/v1/import_refusals 
    # and bulk CSV downloads, but this script fulfills the requirement to scrape and parse the HTML tables.
    
    session = get_session_with_retries()
    
    try:
        html_content = download_reports(session, base_url)
        df = parse_and_clean_data(html_content)
        
        if not df.empty:
            export_data(df)
        else:
            logger.warning("No data extracted. Ensure the site is accessible and the form parameters are correct.")
            
    except Exception as e:
        logger.critical(f"Pipeline failed: {e}")

if __name__ == "__main__":
    main()
```
